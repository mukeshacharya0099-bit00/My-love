// floating hearts
const heartsEl = document.getElementById('hearts');
function spawnHeart(){
  const h = document.createElement('div');
  h.className = 'heart';
  h.textContent = ['💗','💕','💖','🩷'][Math.floor(Math.random()*4)];
  h.style.left = Math.random()*100 + 'vw';
  h.style.setProperty('--drift', (Math.random()*80-40)+'px');
  h.style.animationDuration = (7+Math.random()*6)+'s';
  h.style.fontSize = (1+Math.random()*1.2)+'rem';
  heartsEl.appendChild(h);
  setTimeout(()=>h.remove(), 14000);
}
setInterval(spawnHeart, 500);
for(let i=0;i<6;i++) setTimeout(spawnHeart, i*300);

// confetti
const canvas = document.getElementById('confetti-canvas');
const ctx = canvas.getContext('2d');
function resize(){ canvas.width = innerWidth; canvas.height = innerHeight; }
resize(); addEventListener('resize', resize);
let particles = [];
const colors = ['#f4c95d','#c2185b','#c9a6ff','#ffd9e8','#ffffff'];
function burst(count){
  for(let i=0;i<count;i++){
    particles.push({
      x: canvas.width/2 + (Math.random()-0.5)*canvas.width*0.6,
      y: -20 - Math.random()*100,
      vx: (Math.random()-0.5)*3,
      vy: 2+Math.random()*3,
      size: 4+Math.random()*5,
      color: colors[Math.floor(Math.random()*colors.length)],
      rot: Math.random()*360,
      vr: (Math.random()-0.5)*10
    });
  }
}
function animate(){
  ctx.clearRect(0,0,canvas.width,canvas.height);
  particles.forEach(p=>{
    p.x += p.vx; p.y += p.vy; p.rot += p.vr;
    ctx.save();
    ctx.translate(p.x,p.y); ctx.rotate(p.rot*Math.PI/180);
    ctx.fillStyle = p.color;
    ctx.fillRect(-p.size/2,-p.size/2,p.size,p.size*0.6);
    ctx.restore();
  });
  particles = particles.filter(p=>p.y < canvas.height+30);
  requestAnimationFrame(animate);
}
animate();
window.addEventListener('load', ()=>burst(140));
document.getElementById('confetti-btn').addEventListener('click', ()=>burst(160));

// soft ambient music via Web Audio (no external files needed)
let audioCtx, playing=false, nodes=[];
const musicBtn = document.getElementById('music-toggle');
function startMusic(){
  audioCtx = new (window.AudioContext||window.webkitAudioContext)();
  const master = audioCtx.createGain();
  master.gain.value = 0.06;
  master.connect(audioCtx.destination);
  const notes = [523.25,587.33,659.25,783.99,659.25,587.33]; // soft melody loop
  let t = audioCtx.currentTime;
  function scheduleLoop(startTime){
    notes.forEach((freq,i)=>{
      const osc = audioCtx.createOscillator();
      const g = audioCtx.createGain();
      osc.type = 'sine';
      osc.frequency.value = freq;
      const noteStart = startTime + i*0.9;
      g.gain.setValueAtTime(0, noteStart);
      g.gain.linearRampToValueAtTime(1, noteStart+0.15);
      g.gain.linearRampToValueAtTime(0, noteStart+0.85);
      osc.connect(g); g.connect(master);
      osc.start(noteStart); osc.stop(noteStart+0.9);
      nodes.push(osc);
    });
  }
  const loopLen = notes.length*0.9;
  scheduleLoop(t);
  const interval = setInterval(()=>{ if(playing){ scheduleLoop(audioCtx.currentTime); } else clearInterval(interval); }, loopLen*1000);
}
musicBtn.addEventListener('click', ()=>{
  playing = !playing;
  musicBtn.textContent = playing ? '🔊' : '🔇';
  if(playing){
    if(!audioCtx) startMusic(); else audioCtx.resume();
  } else if(audioCtx){
    audioCtx.suspend();
  }
});
